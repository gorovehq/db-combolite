db-combolite User Manual

A practical, hands-on developer manual for integrating, configuring, modeling, and querying multi-model data within db-combolite applications.

1. Quick Start Guide

db-combolite is delivered as an embeddable C99 library. It requires no separate installation of database daemons, system services, or socket servers. A single database is represented on disk as an archive file (e.g. app_data.cdbarch).

1.1 Embedding & CMake Integration

Add db-combolite to your project's CMakeLists.txt: 

# Include the combolite subdirectory
add_subdirectory(path/to/db-combolite)

# Link your application against the static library
target_link_libraries(my_application PRIVATE combolite)
target_include_directories(my_application PRIVATE path/to/db-combolite/include)

1.2 Memory Sizing & Caller-Owned Buffers

Because db-combolite operates under a strict zero-heap policy, the host application supplies memory buffers at initialization via cdb_db_config. The library provides helper functions to calculate exact buffer dimensions:

Buffer Field Calculation Helper Purpose
doc_buffers cdb_doc_buffers_size(page_size, doc_max) Working memory for document revisions, JSON patches, and nibblet packing.
fts_buffers cdb_fts_buffers_size(page_size) Inverted index postings and BM25 term dictionary working pages.
table_buffers cdb_table_buffers_size(page_size) Row object decoding, constraint verification, and secondary index maintenance.
vec_buffers 3 * page_size Vector packing, block indexing, and distance metric calculations.
vec_work cdb_vec_work_size(k_neighbours) Priority queue scratchpad for approximate nearest neighbour graph traversals.
ts_buffers 3 * page_size Time-series block compression, delta-of-delta bit packing, and Gorilla XOR.
ts_work cdb_ts_work_size() Staging memory for out-of-order/late arrivals and time-window queries.
slot_mem cdb_db_slot_mem_size(max_open) Pre-allocated cache slots for open stores (tables, documents, vectors).

1.3 Opening & Closing a Database

Opening a database requires initializing the host environment (cdb_hostenv), setting configuration options, and calling cdb_db_open: 

#include <combolite/cdb_common.h>
#include <combolite/cdb_env.h>
#include <cdb_db.h>
#include <cdb_hostenv.h>

cdb_hostenv host;
cdb_env env = cdb_hostenv_make(&host);
cdb_db db;
cdb_db_config cfg;
memset(&cfg, 0, sizeof(cfg));

/* Configure standard 4096-byte pages */
cfg.pager.page_size = 4096;

/* Allocate caller buffers */
uint8_t doc_buf[65536];
cfg.doc_buffers = doc_buf;

/* Open database (creates archive if not present) */
cdb_status s = cdb_db_open(&db, &env, "my_database.cdbarch", &cfg, NULL);
if (s != CDB_OK) {
    fprintf(stderr, "Failed to open database: %s\n", cdb_status_str(s));
    return -1;
}

/* Perform database operations ... */

/* Close and flush companion tracks */
cdb_db_close(&db);

1.4 Transactions & Durability Settings

All mutating operations (documents, tables, links, vectors, and time-series) occur within explicit ACID transactions: 

/* Begin transaction with NORMAL durability (fast, durable at checkpoints) */
s = cdb_tx_begin(&db, CDB_DURABILITY_NORMAL);

/* Stage mutations ... */

/* Commit transaction */
s = cdb_tx_commit(&db);
if (s != CDB_OK) {
    cdb_tx_abort(&db);
}

3.2 Relational Tables CRUD

Rows are encoded as objects keyed by column field IDs. Primary and secondary unique indexes are checked inside the open transaction before journaling. 

/* --- CREATE: Insert a new row --- */
cdb_tx_begin(&db, CDB_DURABILITY_NORMAL);
uint8_t row_buf[256];
cdb_value_writer w;
cdb_value_writer_init(&w, row_buf, sizeof(row_buf));
cdb_value_begin_object(&w);
cdb_value_field_id(&w, col_id);
cdb_value_put_string(&w, "Alice", 5);
cdb_value_end_object(&w);

uint64_t entry_id = 0;
cdb_status s = cdb_table_insert(&db, table_id, row_buf, cdb_value_writer_len(&w), &entry_id);
cdb_tx_commit(&db);

/* --- READ: Find by Primary Key or Scan --- */
int found = 0;
uint64_t found_entry = 0;
s = cdb_table_find(&db, table_id, pk_bytes, pk_len, &found_entry, &found);

uint8_t read_buf[512];
size_t row_len = 0;
s = cdb_table_get(&db, table_id, found_entry, read_buf, sizeof(read_buf), &row_len);

/* --- UPDATE: Partial column update --- */
cdb_tx_begin(&db, CDB_DURABILITY_NORMAL);
uint8_t update_buf[128];
cdb_value_writer uw;
cdb_value_writer_init(&uw, update_buf, sizeof(update_buf));
cdb_value_begin_object(&uw);
cdb_value_field_id(&uw, score_col_id);
cdb_value_put_f64(&uw, 99.5);
cdb_value_end_object(&uw);

s = cdb_table_update(&db, table_id, found_entry, update_buf, cdb_value_writer_len(&uw));
cdb_tx_commit(&db);

/* --- DELETE: Remove row by entry ID --- */
cdb_tx_begin(&db, CDB_DURABILITY_NORMAL);
s = cdb_table_delete(&db, table_id, found_entry);
cdb_tx_commit(&db);

3.3 Document Pools CRUD

Documents maintain append-only revision histories. Deletions write CDB_REV_TOMBSTONE revisions, enabling historical time-travel queries. 

/* --- CREATE: Insert document --- */
cdb_tx_begin(&db, CDB_DURABILITY_NORMAL);
uint64_t doc_entry = 0;
s = cdb_doc_insert(&db, pool_id, "Article summary", 15, json_buf, json_len, &doc_entry);
cdb_tx_commit(&db);

/* --- READ: Read head revision --- */
uint8_t doc_out[4096];
size_t doc_len = 0;
cdb_doc_info info;
s = cdb_doc_get(&db, pool_id, doc_entry, 0 /* 0 = latest */, doc_out, sizeof(doc_out), &doc_len, &info);

/* --- UPDATE: Apply JSON patch --- */
cdb_tx_begin(&db, CDB_DURABILITY_NORMAL);
/* Patch operation: [CDB_PATCH_SET, [path...], new_val] */
s = cdb_doc_patch(&db, pool_id, doc_entry, patch_bytes, patch_len);
cdb_tx_commit(&db);

/* --- DELETE: Tombstone document --- */
cdb_tx_begin(&db, CDB_DURABILITY_NORMAL);
s = cdb_doc_delete(&db, pool_id, doc_entry);
cdb_tx_commit(&db);

/* --- CREATE: Stream write blob --- */

s = cdb_db_blob_write_begin(&db, store_id, scratch, &bw);
while ((n = fread(chunk, 1, sizeof(chunk), fp)) > 0) {
    cdb_db_blob_write_chunk(&bw, chunk, n);
}

/* --- READ: Stream read blob --- */
cdb_blob_reader br;
uint8_t page_buf[4096];
s = cdb_db_blob_read_begin(&db, store_id, &ref, page_buf, 1 /* verify hash */, &br);
while ((s = cdb_db_blob_read_chunk(&br, dest_buf, sizeof(dest_buf), &out_len)) == CDB_OK && out_len > 0) {
    fwrite(dest_buf, 1, out_len, out_fp);
}
cdb_db_blob_read_end(&br);

/* --- DELETE: Free blob pages --- */
s = cdb_db_blob_free(&db, store_id, &ref, page_buf);

3.8.2 Atomic Multi-Model Ingestion Procedure (Worked Example)

This stored procedure demonstrates cross-model atomic CRUD: inserting metadata into a relational table, storing passages in a document pool, indexing dense vector embeddings, searching nearest neighbors, establishing graph links, and appending telemetry metrics within a single ACID transaction: 

PROCEDURE ingest_gutenberg_book(
    $ebook_id    INT,
    $title       STRING,
    $author      STRING,
    $language    STRING,
    $passages    ARRAY,   -- Array of { seq: INT, text: STRING, vector: VECTOR(F32, 384) }
    $book_vector VECTOR(F32, 384),
    $client_id   STRING
)
BEGIN TRANSACTION;
    LET $start_time = clock_now();

    -- 1. [CREATE] Relational Table Insert with PK check
    INSERT INTO TABLE catalog (
        ebook_id, title, author, language, downloads, indexed_at
    ) VALUES (
        $ebook_id, $title, $author, $language, 0, $start_time
    );

    -- 2. [CREATE] Document Pool + Full-Text Search and Vector Ingestion
    FOR $p IN $passages DO
        -- Insert document revision (triggers BM25 inverted index update)
        LET $doc_ref = INSERT INTO POOL passages {
            ebook_id: $ebook_id,
            seq:      $p.seq,
            text:     $p.text
        };

        -- Insert vector embedding linked to source passage document
        INSERT INTO VECTOR passage_vec ($vector, source_ref)
        VALUES ($p.vector, $doc_ref);
    END FOR;

    -- 3. [READ] Search Nearest Neighbours using VectorSearch operator
    LET $similar_books = (
        SELECT $target_ref
        FROM VectorSearch(
            Scan("book_vec"),
            Field("$vector"),
            near: $book_vector,
            k: 5,
            metric: "cosine"
        )
        PROJECT {
            $target_ref: Field("source_ref"),
            similarity:  Field("$score")
        }
        FILTER Field("similarity") >= 0.82
           AND Field("$target_ref") != REF(catalog, $ebook_id)
    );

    -- 4. [CREATE] Establish Directed Graph Links with Cosine Weights
    FOR $sim IN $similar_books DO
        LINK REF(catalog, $ebook_id)
          TO $sim.$target_ref
        TYPE "similar_to"
        PROPERTIES { score: $sim.similarity };
    END FOR;

    -- 5. [CREATE] Append Time Series Telemetry Metrics
    LET $duration_ms = (clock_now() - $start_time) / 1000000;
    APPEND TO TIMESERIES ingest_metrics (
        timestamp,
        tags: {
            ebook_id: $ebook_id,
            client:   $client_id
        },
        values: {
            passage_count: $passages.length,
            duration_ms:   $duration_ms,
            status_code:   200
        }
    );

COMMIT;

3.8.3 Stored Procedure Bytecode Mutation Opcodes (Spec 14.20)

At compile time, CMBQL mutation statements compile to CDB_BC_CALL_MUTATION (opcode 0x08): 

/* CALL_MUTATION instruction operands:
 * [0x08, dst_reg, mut_op, store_reg, entry_reg, payload_reg]
 *
 * Supported Mutation Operations (mut_op):
 *   CDB_MUT_DOC_INSERT   (0x01) - Ingests document revision & triggers BM25
 *   CDB_MUT_DOC_REPLACE  (0x02) - Overwrites document body revision
 *   CDB_MUT_TABLE_INSERT (0x03) - Inserts row with unique constraint enforcement
 *   CDB_MUT_TABLE_UPDATE (0x04) - Partially updates row columns
 *   CDB_MUT_LINK_ADD     (0x05) - Connects entities with cardinality checks
 *   CDB_MUT_TS_APPEND    (0x06) - Appends bit-compressed telemetry point
 */

4. Document Pools & BM25 Full-Text Search

Document pools store flexible JSON objects with linear revision chains and integrated text search.

Inserting Documents & Delta Patches

/* 1. Build document payload */
uint8_t doc_bytes[1024];
cdb_value_writer w;
cdb_value_writer_init(&w, doc_bytes, sizeof(doc_bytes));
cdb_value_begin_object(&w);
cdb_value_field(&w, "title", 5);
cdb_value_put_string(&w, "Clean Architecture", 18);
cdb_value_field(&w, "author", 6);
cdb_value_put_string(&w, "Robert Martin", 13);
cdb_value_end_object(&w);

size_t doc_len = cdb_value_writer_len(&w);

/* 2. Insert into document pool */
uint64_t doc_entry_id = 0;
cdb_tx_begin(&db, CDB_DURABILITY_NORMAL);
cdb_doc_insert(&db, pool_id, doc_bytes, doc_len, NULL, 0, &doc_entry_id);
cdb_tx_commit(&db);

BM25 Full-Text Search

Configure indexed text paths using cdb_fts_config. Once indexed, query with cdb_fts_search: 

cdb_fts_hit hits[10];
size_t hit_count = 0;

/* Searches for 'Architecture' across indexed text fields */
cdb_fts_search(&db, pool_id, "Architecture", 12, hits, 10, &hit_count);
for (size_t i = 0; i < hit_count; i++) {
    printf("Match: Doc ID %llu, BM25 Score: %.4f\n", 
           (unsigned long long)hits[i].entry, hits[i].score);
}

9. CMBQL Query Language Guide

CMBQL supports both declarative SQL syntax and pipeline (Flux) syntax. Both compile directly into the identical binary operator wire tree (CMBQ).

Declarative SQL Syntax Example

SELECT title, author, score
FROM accounts
WHERE score > 85.0
ORDER BY score DESC
LIMIT 10;

Pipeline (Flux) Multi-Model Fusion Example

scan("documents")

13. Mathematical, Scientific & DSP Computing Guide

db-combolite includes an embedded 110-function scientific computing engine (cdb_math.h). It executes mathematics directly inside database queries and stored procedures without heap allocation or external math libraries.

13.1 Using Math Functions in Declarative CMBQL Queries

Every built-in math function is directly available inside CMBQL SELECT projection expressions, WHERE filtering clauses, and ORDER BY sort specifications: 

-- 1. Proximity search: find vehicles within 5000 meters of San Francisco
SELECT id, driver, haversine(lat, lon, 37.7749, -122.4194) AS dist_meters
FROM vehicles
WHERE haversine(lat, lon, 37.7749, -122.4194) < 5000.0
ORDER BY dist_meters ASC;

-- 2. Neural activation: compute sigmoid probability and GELU activation
SELECT user_id, sigmoid(score) AS win_probability, gelu(raw_weight) AS activated_weight

13.3 Worked Recipe 2: Software-Defined Radio (SDR) & DSP Spectral Analysis

Process raw I/Q baseband samples, apply a Hann window, compute in-place Fast Fourier Transform, and inspect spectral magnitude: 

/* C99 Recipe: SDR baseband FFT power spectral density */
#define FFT_SIZE 1024
float signal[FFT_SIZE];
/* ... populate signal with raw ADC/SDR baseband audio samples ... */

/* Step 1: Pre-compute and apply Hann window to minimize spectral leakage */
float window[FFT_SIZE];
cdb_math_window_hann(window, FFT_SIZE);
for (size_t i = 0; i < FFT_SIZE; i++) {
    signal[i] *= window[i];
}

/* Step 2: Compute single-sided magnitude spectrum (zero-heap) */
float work_r[FFT_SIZE], work_i[FFT_SIZE];
float magnitude[FFT_SIZE / 2 + 1];

cdb_status s = cdb_math_fft_magnitude(signal, FFT_SIZE, work_r, work_i, magnitude);
if (s == CDB_OK) {
    /* Find peak frequency bin */
    float peak_mag = 0.0f;
    size_t peak_bin = 0;
    for (size_t k = 0; k < FFT_SIZE / 2 + 1; k++) {
        if (magnitude[k] > peak_mag) {
            peak_mag = magnitude[k];
            peak_bin = k;
        }
    }
    printf("Dominant tone detected at FFT bin %zu, magnitude: %.2f (%.1f dBFS)\n", 
           peak_bin, peak_mag, cdb_math_dbfs(peak_mag * peak_mag));
}

/* Step 3: Demodulate FM audio from I/Q stream */
double audio_sample = cdb_math_iq_demod_fm(i_curr, q_curr, i_prev, q_prev);
